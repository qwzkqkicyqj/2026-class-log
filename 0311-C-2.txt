<: 미만
>: 초과
<=: 이하
>=: 이상
==: 같다
!=: 다르다

#include <stdio.h>
int main(){
    int num;
    
    printf("정수를 입력하세요.: ");
    scanf("%d", &num); //정수 입력
    
    //조건문
    if (num % 2 == 0){ //입력받은 정수를 2로 나누었을 때의 나머지 값을 기준으로 나눔.
        printf("짝수\n"); //나머지가 0인 경우
        printf("2로 나눈 나머지가 0\n");
    } else { //앞서 사용한 if문에 해당되지 않는 경우 else문 실행.
        printf("홀수\n"); //나머지가 1인 경우
        printf("2로 나눈 나머지가 1\n");
    } //if문에 들어가는 명령어가 2줄 이상인 경우 중괄호({})를 사용
    return 0;
}
